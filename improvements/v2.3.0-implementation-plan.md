#!/bin/bash
# Common functions used across Claude Context System commands
# Version: 2.3.0

# Find context directory (used in 8+ commands)
find_context_dir() {
  for dir in "context" "../context" "../../context"; do
    if [ -d "$dir" ]; then
      echo "$dir"
      return 0
    fi
  done
  return 1
}

# Robust download with retry (fixes Issue #2)
download_with_retry() {
  local url=$1
  local output=$2
  local attempts=3
  local timeout=10

  for i in $(seq 1 $attempts); do
    if curl --connect-timeout $timeout --max-time 30 -sL "$url" -o "$output" 2>/dev/null; then
      return 0
    fi
    echo "⚠️  Attempt $i/$attempts failed, retrying..." >&2
    sleep $((i * 2))
  done

  echo "❌ Failed to download after $attempts attempts" >&2
  echo "   URL: $url" >&2
  echo "   Check your internet connection" >&2
  return 1
}

# Validate user input (fixes Issue #4)
validate_input() {
  local input=$1
  local pattern=${2:-'^[a-zA-Z0-9_-]+$'}
  local max_length=${3:-50}

  if [ ${#input} -gt $max_length ]; then
    echo "❌ Error: Input too long (max $max_length characters)" >&2
    return 1
  fi

  if [[ ! "$input" =~ $pattern ]]; then
    echo "❌ Error: Invalid input format" >&2
    echo "   Allowed: letters, numbers, dashes, underscores" >&2
    return 1
  fi

  return 0
}

# Efficient file scanning (fixes Issue #1)
find_md_files() {
  local max_depth=${1:-3}
  find . -maxdepth $max_depth -name "*.md" \
    -not -path "*/node_modules/*" \
    -not -path "*/.git/*" \
    -not -path "*/dist/*" \
    -not -path "*/build/*" \
    -not -path "*/.next/*" \
    -not -path "*/out/*" \
    2>/dev/null
}

# Progress indicator (fixes Issue #10)
show_progress() {
  local message=$1
  local step=$2
  local total=$3

  if [ -n "$step" ] && [ -n "$total" ]; then
    echo "⏳ [$step/$total] $message..."
  else
    echo "⏳ $message..."
  fi
}

# Error message helper (fixes Issue #9)
show_error() {
  local error_code=$1
  local message=$2
  shift 2
  local suggestions=("$@")

  echo "" >&2
  echo "❌ Error ($error_code): $message" >&2
  echo "" >&2

  if [ ${#suggestions[@]} -gt 0 ]; then
    echo "This usually means:" >&2
    for suggestion in "${suggestions[@]}"; do
      echo "  • $suggestion" >&2
    done
    echo "" >&2
  fi
}

# Cache operations (Improvement #4)
CACHE_DIR=".claude/.cache"
CACHE_TTL=60

get_cached() {
  local key=$1
  local file="$CACHE_DIR/$key"

  if [ -f "$file" ]; then
    local age=$(($(date +%s) - $(stat -f %m "$file" 2>/dev/null || stat -c %Y "$file")))
    if [ $age -lt $CACHE_TTL ]; then
      cat "$file"
      return 0
    fi
  fi
  return 1
}

set_cached() {
  local key=$1
  local value=$2
  mkdir -p "$CACHE_DIR"
  echo "$value" > "$CACHE_DIR/$key"
}

clear_cache() {
  rm -rf "$CACHE_DIR"
  echo "✅ Cache cleared"
}

# Configuration repository URL (fixes Issue #7)
get_repo_url() {
  local config_file="context/.context-config.json"

  if [ -f "$config_file" ]; then
    local custom_repo=$(grep -o '"repositoryUrl"[[:space:]]*:[[:space:]]*"[^"]*"' "$config_file" | cut -d'"' -f4)
    if [ -n "$custom_repo" ]; then
      echo "$custom_repo"
      return
    fi
  fi

  # Default
  echo "https://github.com/rexkirshner/claude-context-system"
}

get_raw_url() {
  local repo=$(get_repo_url)
  echo "${repo/github.com/raw.githubusercontent.com}/main"
}

# Version from single source of truth
get_system_version() {
  if [ -f "VERSION" ]; then
    cat VERSION
  else
    grep -m 1 '"version":' context/.context-config.json 2>/dev/null | cut -d'"' -f4 || echo "unknown"
  fi
}

# Exit codes (Improvement #10)
EXIT_SUCCESS=0
EXIT_ERROR=1
EXIT_MISUSE=2
EXIT_NOT_FOUND=3
EXIT_NETWORK=4
EXIT_PERMISSION=5
EXIT_VALIDATION=6

#### 3.5: Improvement #5 - Auto-Update Notifications

**Add to:** All commands (in common-functions.sh)

**Implementation:**
```bash
check_for_updates() {
  local check_file=".claude/.last-update-check"
  local check_interval=86400  # 24 hours

  # Check once per day
  if [ -f "$check_file" ]; then
    local age=$(($(date +%s) - $(stat -f %m "$check_file" 2>/dev/null || stat -c %Y "$check_file")))
    if [ $age -lt $check_interval ]; then
      return 0
    fi
  fi

  # Get latest version from GitHub
  local repo_url=$(get_repo_url)
  local latest=$(curl -s --connect-timeout 5 --max-time 10 \
    "https://api.github.com/repos/${repo_url#https://github.com/}/releases/latest" \
    2>/dev/null | grep '"tag_name"' | cut -d'"' -f4 | sed 's/^v//')

  local current=$(get_system_version)

  if [ -n "$latest" ] && [ "$latest" != "$current" ]; then
    echo "" >&2
    echo "💡 Update available: v$current → v$latest" >&2
    echo "   Run /update-context-system to upgrade" >&2
    echo "" >&2
  fi

  touch "$check_file"
}

# Call at start of each command (non-blocking)
check_for_updates &
```

#### 4.4: Fix Issue #8 - Download Validation

**Add to:** download_with_retry() in common-functions.sh

```bash
download_with_retry() {
  # ... existing retry logic ...

  # Validate downloaded file
  if [ -f "$output" ]; then
    # Check it's not an HTML error page
    if head -1 "$output" | grep -qi "<!DOCTYPE\|<html"; then
      echo "❌ Downloaded file appears to be HTML (likely an error page)" >&2
      rm -f "$output"
      return 1
    fi

    # Check file size is reasonable (> 10 bytes, < 10MB)
    local size=$(wc -c < "$output")
    if [ $size -lt 10 ]; then
      echo "❌ Downloaded file too small ($size bytes)" >&2
      rm -f "$output"
      return 1
    fi

    if [ $size -gt 10485760 ]; then
      echo "⚠️  Warning: Downloaded file is large ($size bytes)" >&2
    fi

    return 0
  fi

  return 1
}
```
